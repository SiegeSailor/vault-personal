- `GROUP BY`: Divide the result into groups with given column names
- `HAVING`: Return the group if the specified condition is True
- `7-32` and `7-33` explains the importance of the execution order (`WHERE` comes before `GROUP BY`, so the result from `HAVING` is affected)
- `EXISTS` vs `IN`: `EXISTS` is faster, use `IN` only when there is something to check
- `WITH`: Create a temporary table with a query to make a more cleaning query

## In Class Practice

```sql
USE company;

-- 1
SELECT fname, lname
FROM employee
WHERE fname LIKE 'A%' OR lname LIKE '__z%';

-- 2
SELECT ssn, fname, lname
FROM employee
WHERE ssn NOT IN (SELECT works_on.essn FROM works_on, project WHERE works_on.pno = project.pnumber AND pname = 'ProductX');

-- 3
SELECT dname
FROM department, dept_locations, project
WHERE department.dnumber = dept_locations.dnumber
AND department.dnumber = project.dnum
AND dlocation = 'Houston'
AND plocation = 'Bellaire';
-- or
SELECT dname
FROM (department JOIN dept_locations ON department.dnumber = dept_locations.dnumber) JOIN project ON department.dnumber = project.dnum
WHERE dlocation = 'Houston' AND plocation = 'Bellaire';

-- 4
SELECT dnumber, COUNT(*)
FROM department, employee
WHERE dnumber = dno
GROUP BY dno;

-- 5
SELECT dname, COUNT(*)
FROM department, employee
WHERE dnumber = dno
GROUP BY dno;

-- 6
SELECT pnumber, pname
FROM project, works_on
WHERE pno = pnumber
GROUP BY pno
HAVING COUNT(*) >= 3;

-- 7
-- Below is showing not supervisors
SELECT e.fname, e.lname
FROM employee AS e
WHERE NOT EXISTS (SELECT * FROM employee WHERE superssn = e.ssn);
-- Below is correct, showing not managers
SELECT fname, lname
FROM employee
WHERE ssn NOT IN (SELECT mgrssn FROM department);

-- 8
SELECT fname, minit, lname, dependent_name

-- 9

FROM department
WHERE NOT EXISTS (SELECT * FROM dept_locations WHERE department.dnumber = dept_locations.dnumber AND dlocation = 'Houston');
```
